    - id : int
    - titulo : String

    - multaPorDia : double
    - contadorPrestamos : int
    - multaAcumulada : double
    + getId() : int
    + getTitulo() : String
    + isDisponible() : boolean
    + setDisponible(boolean)
    + getDiasMaximosPrestamo() : int
    + getMultaPorDia() : double
    + getContadorPrestamos() : int
    + getMultaAcumulada() : double
    + setMultaAcumulada(double)
    + incrementarContadorPrestamos()
    + mostrarInformacion()
    + calcularMulta(int) : double
}

class Libro <<Libro>> {
    - paginas : int
    - autor : String
    + Libro(int, String, int, String)
    + mostrarInformacion()
    + calcularMulta(int) : double
}

class Revista <<Revista>> {
    - periodicidad : String
    - numeroEdicion : int
    + Revista(int, String, String, int)
    + mostrarInformacion()
    + calcularMulta(int) : double
}

    - duracionMinutos : int

    + Pelicula(int, String, int, String)
    + mostrarInformacion()
    + calcularMulta(int) : double
}

    - id : int
    - nombre : String

    + getNombre() : String

    + getMultaAcumulada() : double
    + agregarMulta(double)
    + mostrarDatos()
}

    - material : MaterialBibliografico
    - fechaPrestamo : LocalDate
    - fechaDevolucion : LocalDate
    - multaCalculada : double
    - devuelto : boolean
    + Prestamo(Usuario, MaterialBibliografico, LocalDate)
    + devolver(LocalDate)
    + getUsuario() : Usuario
    + getMaterial() : MaterialBibliografico
    + getFechaPrestamo() : LocalDate
    + getFechaDevolucion() : LocalDate
    + getMultaCalculada() : double
